#-------------------------------------------qt依赖----------------------------------------
add_definitions(-DQT_NO_DEBUG_OUTPUT) #禁用qDebug 输出
# 开启CMake的自动包含当前目录的功能，以确保能够找到当前源代码中的头文件
set(CMAKE_INCLUDE_CURRENT_DIR ON)#设置工程包含当前目录，使用*.ui文件时，需要加上这句，否则找不到头文件
# 开启CMake的自动MOC功能，用于处理Qt的元对象编译（Meta-Object Compiler）
set(CMAKE_AUTOMOC ON)#打开全局moc
# 设置CMake自动处理UI文件
# AUTOUIC：自动处理Qt Designer的UI文件，将它们转换为对应的C++代码
set(CMAKE_AUTOUIC ON)#打开全局uic
# 设置CMake自动处理资源文件
# AUTORCC：自动处理Qt的资源文件（Resource Compiler）
set(CMAKE_AUTORCC ON)#打开全局rcc
find_package(Qt5 REQUIRED Widgets )


#--------------------------------------------end------------------------------------------
